Main commands

    # odometry position control
    robot.forward(m)    move forward m [m]
    robot.backward(m)   move backward m [m]
    robot.left(r)       turn left r [deg]
    robot.right(r)      turn right r [deg]

    # odometry velocity control
    robot.setSpeed(lx,az,time,stopend=False)
                  lx: linear velocity [m/s]
                  az: angular velocity [rad/s]
                  time: time [s]
                  stopend: stop after motion 

    # head position
    robot.pan(deg): positive left
    robot.tilt(deg): positive up

    # arms position
    robot.left_arm(deg): positive ahead     
    robot.right_arm(deg): positive ahead

    # text to speech
    robot.say(sentence, language): language = 'en'|'it'

    # speech to text
    sentence = robot.asr(timeout=5): sentence said by human (or None after timeout)

    # odometry pose [m, deg])
    x,y,th_deg = robot.get_pose()

    # joints position [deg]
    deg = robot.get_pan_pos()
    deg = robot.get_tilt_pos()
    deg = robot.get_left_arm_pos()
    deg = robot.get_right_arm_pos()

    # obstacle distance 
    m = robot.obstacle_distance(deg=0)
    
    # image
    robot.get_image()
