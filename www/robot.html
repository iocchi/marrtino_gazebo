Main commands

    # position control
    robot.forward(m)    move forward m [m]
    robot.backward(m)   move backward m [m]
    robot.left(r)       turn left r [deg]
    robot.right(r)      turn right r [deg]

    # robot pose [m, deg]
    x,y,th_deg = robot.get_pose(frame='odom'/'gt')

    # goto x,y with optional theta_deg final orientation
    robot.goto(x, y, [theta_deg])  

    # odometry velocity control
    robot.setSpeed(lx,az,time,stopend=False)
                  lx: linear velocity [m/s]
                  az: angular velocity [rad/s]
                  time: time [s]
                  stopend: stop after motion 

    # head position
    robot.pan(deg): positive left
    robot.tilt(deg): positive up

    # arms position
    robot.left_arm(deg): positive ahead     
    robot.right_arm(deg): positive ahead

    # text to speech
    robot.say(sentence, language): language = 'en'|'it'

    # speech to text
    sentence = robot.asr(timeout=5): sentence said by human (or None after timeout)


    # joints position [deg]
    deg = robot.get_pan_pos()
    deg = robot.get_tilt_pos()
    deg = robot.get_left_arm_pos()
    deg = robot.get_right_arm_pos()

    # obstacle distance 
    m = robot.obstacle_distance(deg=0)
    
    # image
    robot.get_image()

    # faces
    robot.emotion(value)   # normal, happy, sad, angry, surprised, sings

    # AI functionalities (need OpenAI/Gemini key)
    
    robot.ai_say(prompt)    # robot says the output of LLM prompt
    robot.ai_code(prompt)   # robot executes the output of LLM prompt
    robot.ai_vision(prompt) # robot takes an image of the current scene and says the output of VLM prompt on such an image
    
    a = robot.ai.chat(prompt)   # returns the string output of LLM prompt
    p = robot.ai.code(prompt)   # returns the code output of LLM prompt
    exec(p)                     # executes a program
    d = robot.ai.vision(image, prompt)    # returns the string output of VLM prompt on the given image
    a = robot.ai.query(description, prompt)    # returns the output of LLM prompt on the description
    robot.ai.memorize(content)    # memorize content in the vector DB
    a = robot.ai.retrieve(prompt, max_dist=2.0, n_results=3)   # returns output of prompt retrieving documents from the vector DB
    robot.ai.clear_memory()       # clear the all memory
    m = robot.ai.list_memory()    # string with items in the current memory
    robot.ai.forget(ids)          # remove from memory content with specified ids
